Založ si zdarma projekt na supabase.com, spusť SQL níže a sem vlož adresu projektu a anon klíč. Klíč service_role sem nikdy nedávej.

Zobrazit SQL pro založení tabulky 
create table planner_data (
  user_id uuid primary key references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);

alter table planner_data enable row level security;

create policy "vlastni radek" on planner_data
  for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
